// auth.interceptor.ts:32 - ก่อนแก้ (มีปัญหา)
catchError((error) => {
  if (error instanceof HttpErrorResponse && error.status === 401) {
    return this.handle401Error(req, next);
  }

})

// auth.interceptor.ts:32 - หลังแก้ (ทำงานได้)
catchError((error) => {
  const status = error?.status || error?.statusCode;
  if (status === 401 && !isRefreshRequest) {
    return this.handle401Error(req, next);
  }
  return throwError(() => error);
})

เปลี่ยนจาก instanceof HttpErrorResponse เป็น error?.status เพื่อรองรับ plain object ที่ ErrorInterceptor แปลงมา

วิธียืนยัน

รันคำสั่งนี้เพื่อยืนยันว่า proxy ทำงาน:

curl -X POST http://localhost:4200/api/auth/login -H "Content-Type: application/json" -d '{"username":"test","password":"test"}'

ผลลัพธ์ที่คาดหวัง:

  • ถ้าได้ 422 → Proxy ทำงานแล้ว ปัญหาอยู่ที่ backend
  • ถ้าได้ 200 → Login สำเร็จ
  • ถ้าได้ CORS error → ยังมีปัญหา proxy config
