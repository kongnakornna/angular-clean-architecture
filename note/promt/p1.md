**`data/repositories/translation.repository.impl.ts`**

```typescript
// features/i18n/data/repositories/translation.repository.impl.ts
import { Injectable, inject } from '@angular/core';
import { TranslateService } from '@ngx-translate/core';
import { Observable, of } from 'rxjs';
import { ITranslationRepository } from '../../domain/repositories/translation.repository';
import { Translation, SupportedLanguage, LanguageOption } from '../../domain/entities/translation.entity';
import { TranslationApiDataSource } from '../datasources/translation.api.datasource';

@Injectable({ providedIn: 'root' })
export class TranslationRepositoryImpl implements ITranslationRepository {
  private translate = inject(TranslateService);
  private api = inject(TranslationApiDataSource);

  private readonly STORAGE_KEY = 'preferred_language';
  private currentLang: SupportedLanguage = 'th';

  private languageOptions: LanguageOption[] = [
    { code: 'th', name: 'ไทย', flag: '🇹🇭' },
    { code: 'en', name: 'English', flag: '🇬🇧' },
  ];

  constructor() {
    // โหลดภาษาที่บันทึกไว้
    const saved = localStorage.getItem(this.STORAGE_KEY) as SupportedLanguage | null;
    if (saved && this.languageOptions.some(l => l.code === saved)) {
      this.currentLang = saved;
    } else {
      // ใช้ภาษาของ Browser หรือค่าเริ่มต้น
      const browserLang = this.translate.getBrowserLang() as SupportedLanguage;
      this.currentLang = browserLang?.match(/th|en/) ? browserLang : 'th';
    }
    this.translate.use(this.currentLang);
  }

  getTranslations(lang: SupportedLanguage): Observable<Translation[]> {
    // ใช้ TranslateService.getTranslation(lang) หรือเรียก API
    return this.api.getTranslations(lang);
  }

  getAvailableLanguages(): Observable<LanguageOption[]> {
    return of(this.languageOptions);
  }

  setLanguage(lang: SupportedLanguage): void {
    this.currentLang = lang;
    localStorage.setItem(this.STORAGE_KEY, lang);
    this.translate.use(lang);
  }

  getCurrentLanguage(): SupportedLanguage {
    return this.currentLang;
  }

  instant(key: string): string {
    return this.translate.instant(key);
  }
}
```

#### 3.4 Presentation Layer

**`presentation/pipes/translate.pipe.ts`**

```typescript
// features/i18n/presentation/pipes/translate.pipe.ts
import { Pipe, PipeTransform, inject, OnDestroy } from '@angular/core';
import { TranslateService } from '@ngx-translate/core';
import { Subscription } from 'rxjs';

@Pipe({
  name: 'appTranslate',
  standalone: true,
  pure: false // ทำให้ pipe อัปเดตเมื่อภาษาเปลี่ยน
})
export class TranslatePipe implements PipeTransform, OnDestroy {
  private translate = inject(TranslateService);
  private langChangeSubscription: Subscription;

  private value: string = '';
  private lastKey: string = '';

  constructor() {
    // อัปเดตเมื่อภาษาเปลี่ยน
    this.langChangeSubscription = this.translate.onLangChange.subscribe(() => {
      if (this.lastKey) {
        this.value = this.translate.instant(this.lastKey);
      }
    });
  }

  transform(key: string): string {
    if (key !== this.lastKey) {
      this.lastKey = key;
      this.value = this.translate.instant(key);
    }
    return this.value;
  }

  ngOnDestroy(): void {
    this.langChangeSubscription?.unsubscribe();
  }
}
```

**`presentation/components/language-selector/language-selector.component.ts`**

```typescript
// features/i18n/presentation/components/language-selector/language-selector.component.ts
import { Component, inject, signal, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ITablerIcon } from 'ngx-tabler-icons';
import { SetLanguageUseCase } from '../../../domain/use-cases/set-language.use-case';
import { GetAvailableLanguagesUseCase } from '../../../domain/use-cases/get-available-languages.use-case';
import { GetCurrentLanguageUseCase } from '../../../domain/use-cases/get-current-language.use-case';
import { LanguageOption } from '../../../domain/entities/translation.entity';

@Component({
  selector: 'app-language-selector',
  standalone: true,
  imports: [CommonModule, ITablerIcon],
  template: `
    <div class="nav-item dropdown">
      <a class="nav-link dropdown-toggle" data-bs-toggle="dropdown" role="button">
        <i-tabler-icon name="language" />
        <span class="d-none d-md-inline ms-1">
          {{ currentLanguage()?.flag }} {{ currentLanguage()?.name }}
        </span>
      </a>
      <div class="dropdown-menu dropdown-menu-end">
        @for (lang of languages(); track lang.code) {
          <a class="dropdown-item"
             [class.active]="currentLanguage()?.code === lang.code"
             (click)="setLanguage(lang.code)">
            <i-tabler-icon *ngIf="currentLanguage()?.code === lang.code" name="check" class="dropdown-icon" />
            {{ lang.flag }} {{ lang.name }}
          </a>
        }
      </div>
    </div>
  `,
  styles: [`
    .dropdown-icon {
      margin-right: 0.5rem;
    }
  `]
})
export class LanguageSelectorComponent implements OnInit {
  private setLangUseCase = inject(SetLanguageUseCase);
  private getLangsUseCase = inject(GetAvailableLanguagesUseCase);
  private getCurrentLangUseCase = inject(GetCurrentLanguageUseCase);

  languages = signal<LanguageOption[]>([]);
  currentLanguage = signal<LanguageOption | null>(null);

  ngOnInit() {
    this.getLangsUseCase.execute().subscribe(langs => {
      this.languages.set(langs);
      const currentCode = this.getCurrentLangUseCase.execute();
      this.currentLanguage.set(langs.find(l => l.code === currentCode) || null);
    });
  }

  setLanguage(code: string) {
    this.setLangUseCase.execute(code as any);
    const lang = this.languages().find(l => l.code === code);
    if (lang) {
      this.currentLanguage.set(lang);
    }
  }
}
```

## 📌 หมายเหตุ

- ใช้ `ngx-tabler-icons` แทน `angular-tabler-icons` เพราะรองรับ Angular 21
- ใช้ `--legacy-peer-deps` เพื่อแก้ปัญหา dependency conflict
- การใช้ Clean Architecture ช่วยให้การเปลี่ยนแหล่งข้อมูล (API → Local JSON) ทำได้ง่าย
- `TranslatePipe` ที่สร้างขึ้นจะอัปเดตค่าโดยอัตโนมัติเมื่อภาษาเปลี่ยน (pure: false)
